// Check if List is Sorted let arr2 = [1, 2, 3, 4, 5, 6, 7, 8, 9]; let isSorted = true; for (let i = 0; i < arr2.length - 1; i++) { if (arr2[i] > arr2[i + 1]) { isSorted = false; break; } } if (isSorted) { console.log("The list is sorted in ascending order."); } else { console.log("The list is not sorted in ascending order."); }
